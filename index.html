# -*- coding: utf-8 -*- import http.server import json import os import webbrowser from urllib.parse import parse_qs, urlparse import joblib import numpy as np import pandas as pd from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score from sklearn.model_selection import train_test_split PORT = 8080 DATA_PATH = "clean_data.xlsx" MODEL_PATH = "rf_bcr_model.joblib" # 全局数据加载 df = pd.DataFrame() model = None metrics = {} min_max = {} p01_p99 = {} def load_resources(): global df, model, metrics, min_max, p01_p99 if os.path.exists(DATA_PATH): df = pd.read_excel(DATA_PATH) if os.path.exists(MODEL_PATH): loaded_obj = joblib.load(MODEL_PATH) # 兼容处理：如果 joblib 保存的是字典，则提取其中的 'model' if isinstance(loaded_obj, dict): model = loaded_obj.get("model", None) else: model = loaded_obj if not df.empty and model is not None: feature_names = [ "水价", "降雨量", "变异系数", "汇水面积", "总建筑面积", "需水量", "人数", "楼高", "水箱容积", "项目寿命", "电价", "贴现率", ] X = df[feature_names] y = df["BCR"] # 自动计算 min/max 以及 1%-99% 分位数 for col in feature_names: min_max[col] = {"min": float(X[col].min()), "max": float(X[col].max())} p01_p99[col] = { "p01": float(X[col].quantile(0.01)), "p99": float(X[col].quantile(0.99)), } # 计算 3:1 划分 (random_state=42) 测试集指标 X_train, X_test, y_train, y_test = train_test_split( X, y, test_size=0.25, random_state=42 ) y_pred = model.predict(X_test) metrics["r2"] = float(r2_score(y_test, y_pred)) metrics["mae"] = float(mean_absolute_error(y_test, y_pred)) metrics["rmse"] = float(np.sqrt(mean_squared_error(y_test, y_pred))) # 将随机森林树结构转为前端 JS 调用的字典格式 def export_rf_to_dict(rf_model): trees_data = [] for tree in rf_model.estimators_: t_ = tree.tree_ tree_dict = { "children_left": t_.children_left.tolist(), "children_right": t_.children_right.tolist(), "feature": t_.feature.tolist(), "threshold": t_.threshold.tolist(), "value": t_.value[:, 0, 0].tolist(), } trees_data.append(tree_dict) return trees_data HTML_TEMPLATE = """ 屋面雨水回收（RWH）经济性预测工具

模型结果校验面板 (测试集 3:1 划分, random_state=42)

""" class RWHHandler(http.server.BaseHTTPRequestHandler): def do_GET(self): parsed_path = urlparse(self.path) if parsed_path.path == "/": self.send_response(200) self.send_header("Content-Type", "text/html; charset=utf-8") self.end_headers() self.wfile.write(HTML_TEMPLATE.encode("utf-8")) elif parsed_path.path == "/api/init": self.send_response(200) self.send_header("Content-Type", "application/json; charset=utf-8") self.end_headers() trees_data = export_rf_to_dict(model) if model is not None else [] payload = { "min_max": min_max, "p01_p99": p01_p99, "metrics": metrics, "raw_data": df.to_dict(orient="records") if not df.empty else [], "trees": trees_data, } self.wfile.write(json.dumps(payload).encode("utf-8")) else: self.send_response(404) self.end_headers() def log_message(self, format, *args): pass def run_server(): load_resources() server_address = ("", PORT) httpd = http.server.HTTPServer(server_address, RWHHandler) url = f"http://localhost:{PORT}" print(f"[*] 屋面雨水回收经济性预测工具已启动: {url}") webbrowser.open(url) try: httpd.serve_forever() except KeyboardInterrupt: print("\n[*] 服务已关闭") if __name__ == "__main__": run_server()
